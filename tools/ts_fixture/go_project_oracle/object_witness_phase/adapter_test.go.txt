const objectPhaseProviderCommit = "5251b6ba8695702a670862c0d4f2b44dba39bf5e"

type objectPhaseReceipt struct {
	Sequence  uint64           `json:"receipt_sequence"`
	Pass      string           `json:"pass_token"`
	Program   string           `json:"program_token"`
	Checker   string           `json:"checker_token"`
	Phase     string           `json:"phase"`
	Query     string           `json:"query_token"`
	Operation string           `json:"query_operation_token"`
	Witness   string           `json:"witness_handle_token"`
	Result    string           `json:"returned_type_token"`
	Rendered  *phaseOracleNode `json:"rendered_node"`
	Queried   *phaseOracleNode `json:"queried_node"`
	Candidate bool             `json:"object_root_candidate"`
}

type objectPhasePassReport struct {
	Token               string `json:"pass_token"`
	Name                string `json:"name"`
	Program             string `json:"program_token"`
	Checker             string `json:"checker_token"`
	OpenedAfterSnapshot string `json:"opened_after_snapshot"`
	ClosedAfterSnapshot string `json:"closed_after_snapshot"`
	BeginOperation      string `json:"begin_operation"`
	EndOperation        string `json:"end_operation"`
	Closed              bool   `json:"closed"`
	Queries             int    `json:"wrapped_type_queries"`
	Candidates          int    `json:"object_root_candidates"`
	RawReturnsMatch     bool   `json:"raw_observer_returns_match"`
}

type objectPhasePair struct {
	ColdQuery      string                                  `json:"cold_query_token"`
	WarmQuery      string                                  `json:"warm_query_token"`
	ColdWitness    string                                  `json:"cold_witness_handle_token"`
	WarmWitness    string                                  `json:"warm_witness_handle_token"`
	ColdResult     string                                  `json:"cold_returned_type_token"`
	WarmResult     string                                  `json:"warm_returned_type_token"`
	Queried        *phaseOracleNode                        `json:"queried_node"`
	PointerEqual   bool                                    `json:"raw_pointer_equal"`
	BeginOperation string                                  `json:"verification_begin_operation"`
	EndOperation   string                                  `json:"verification_end_operation"`
	Verification   checker.ProjectOracleObjectVerification `json:"provider_verification"`
}

type objectPhaseAllocations struct {
	Types      uint32 `json:"types"`
	Symbols    uint32 `json:"symbols"`
	Signatures uint32 `json:"signatures"`
}

type objectPhaseProof struct {
	Schema                  int                      `json:"schema_version"`
	Scope                   string                   `json:"scope"`
	State                   string                   `json:"state"`
	Reason                  string                   `json:"reason"`
	ProviderCommit          string                   `json:"provider_commit"`
	SourceCommit            string                   `json:"source_commit"`
	BinarySHA256            string                   `json:"binary_sha256"`
	ConfigSHA256            string                   `json:"config_sha256"`
	InputGraphHashes        map[string]string        `json:"input_graph_sha256"`
	HandleTokenScope        string                   `json:"handle_token_scope"`
	Passes                  []*objectPhasePassReport `json:"passes"`
	Receipts                *phaseOracleTrace        `json:"receipts"`
	PairingState            string                   `json:"pairing_state"`
	Pairs                   []objectPhasePair        `json:"pairs"`
	VerificationStates      map[string]int           `json:"verification_states"`
	PrimaryBefore           string                   `json:"primary_snapshot_sha256_before_verification"`
	PrimaryAfter            string                   `json:"primary_snapshot_sha256_after_verification"`
	PrimaryUnchanged        bool                     `json:"primary_snapshot_unchanged"`
	ArtifactsUnchanged      bool                     `json:"artifact_bytes_unchanged_after_verification"`
	RawChecks               phaseOraclePointerChecks `json:"preserved_raw_checks"`
	Outcome                 string                   `json:"preserved_phase_outcome"`
	VerificationBegin       string                   `json:"verification_phase_begin_operation"`
	VerificationEnd         string                   `json:"verification_phase_end_operation"`
	AllocationsBefore       objectPhaseAllocations   `json:"allocations_before_verification"`
	AllocationsAfter        objectPhaseAllocations   `json:"allocations_after_verification"`
	RelationCalls           int                      `json:"relation_calls"`
	IdentityCacheDelta      int                      `json:"identity_cache_delta"`
	DiagnosticCodes         []int32                  `json:"relation_diagnostic_codes"`
	SuggestionCodes         []int32                  `json:"relation_suggestion_codes"`
	FreshDiagnosticEquality *bool                    `json:"fresh_diagnostic_equality"`
	ParityClaimed           bool                     `json:"parity_claimed"`
}

type objectPhaseEntry struct {
	receipt           objectPhaseReceipt
	rendered, queried *ast.Node
	result            *checker.Type
	witness           *checker.ProjectOracleObjectQueryWitness
	expectedQuery     uint64
}

type objectPhasePass struct {
	report   *objectPhasePassReport
	provider *checker.ProjectOracleObjectPass
	program  *phaseOracleProgram
	entries  []*objectPhaseEntry
}

type objectPhaseAdapter struct {
	report        *objectPhaseProof
	recorder      *phaseOracleRecorder
	passes        []*objectPhasePass
	active        *objectPhasePass
	pending       *objectPhaseEntry
	results       map[*checker.Type]string
	witnesses     map[*checker.ProjectOracleObjectQueryWitness]string
	file          *os.File
	writer        *bufio.Writer
	hasher        hash.Hash
	bytes         int
	count         uint64
	boundaryCheck func(*objectPhaseAdapter)
}

var objectPhaseActive *objectPhaseAdapter

func newObjectPhaseAdapter(output, config string, recorder *phaseOracleRecorder, provenance json.RawMessage) *objectPhaseAdapter {
	file, err := os.OpenFile(filepath.Join(output, "object-query-receipts.jsonl"), os.O_CREATE|os.O_EXCL|os.O_WRONLY, 0o600)
	if err != nil {
		panic(err)
	}
	configBytes, err := os.ReadFile(config)
	if err != nil {
		panic(err)
	}
	var build struct {
		Instrumentation struct {
			Source string `json:"rust_sha"`
		} `json:"instrumentation"`
		Executable struct {
			SHA256 string `json:"sha256"`
		} `json:"executable"`
	}
	if err := json.Unmarshal(provenance, &build); err != nil {
		panic(err)
	}
	report := &objectPhaseProof{Schema: 1, Scope: "bounded_object_query_proof_not_project_parity", State: "observing", ProviderCommit: objectPhaseProviderCommit,
		SourceCommit: build.Instrumentation.Source, BinarySHA256: build.Executable.SHA256, ConfigSHA256: projectOracleHash(configBytes),
		InputGraphHashes: map[string]string{}, HandleTokenScope: "adapter_tokens_bound_to_actual_provider_handles_and_returned_type_pointers_in_this_process",
		Passes: []*objectPhasePassReport{}, Pairs: []objectPhasePair{}, VerificationStates: map[string]int{"verified": 0, "incomplete": 0, "rejected": 0},
		DiagnosticCodes: []int32{}, SuggestionCodes: []int32{}}
	hasher := sha256.New()
	return &objectPhaseAdapter{report: report, recorder: recorder, results: map[*checker.Type]string{}, witnesses: map[*checker.ProjectOracleObjectQueryWitness]string{}, file: file, hasher: hasher, writer: bufio.NewWriterSize(io.MultiWriter(file, hasher), 1<<20)}
}

func objectPhaseIsCandidate(node *ast.Node) bool {
	seen := map[*ast.Node]bool{}
	for node != nil && ast.IsParenthesizedExpression(node) {
		if seen[node] {
			return false
		}
		seen[node] = true
		node = node.Expression()
	}
	return node != nil && ast.IsObjectLiteralExpression(node)
}

func objectPhaseBeginPass(p *phaseOracleProgram, recorder *phaseOracleRecorder, name, snapshot string) {
	a := objectPhaseActive
	if a == nil {
		return
	}
	if a.active != nil || a.pending != nil || recorder != a.recorder || len(a.passes) >= 2 {
		panic("object pass order changed")
	}
	c, done := p.program.GetTypeCheckerForFileExclusive(context.Background(), p.program.SourceFiles()[0])
	defer done()
	if c != p.checker {
		panic("object pass checker changed")
	}
	provider, err := c.ProjectOracleBeginObjectPass()
	if err != nil {
		panic(err)
	}
	report := &objectPhasePassReport{Token: fmt.Sprintf("object-pass-%d", len(a.passes)+1), Name: name, Program: p.report.Token, Checker: p.report.Checker, OpenedAfterSnapshot: snapshot}
	report.BeginOperation = recorder.event(phaseOracleEvent{Program: report.Program, Checker: report.Checker, Phase: name, Operation: "object_provider_pass_opened"})
	pass := &objectPhasePass{report: report, provider: provider, program: p}
	a.active = pass
	a.passes = append(a.passes, pass)
	a.report.Passes = append(a.report.Passes, report)
	objectPhaseQueryOverride = a.query
	objectPhaseQueryReturnObserver = a.returned
	if a.boundaryCheck != nil {
		a.boundaryCheck(a)
	}
}

// This replaces exactly one walker call. Only the provider performs the query.
func (a *objectPhaseAdapter) query(c *checker.Checker, rendered, queried *ast.Node) *checker.Type {
	pass := a.active
	if pass == nil || a.pending != nil || c != pass.program.checker {
		panic("object query is outside its active checker pass")
	}
	result, witness, err := c.ProjectOracleObjectQuery(pass.provider, queried)
	if err != nil {
		panic(err)
	}
	if witness == nil || a.witnesses[witness] != "" {
		panic("object provider did not issue a unique witness handle")
	}
	a.count++
	witnessToken := fmt.Sprintf("object-witness-%08d", a.count)
	a.witnesses[witness] = witnessToken
	resultToken := "nil"
	if result != nil {
		resultToken = a.results[result]
		if resultToken == "" {
			resultToken = fmt.Sprintf("returned-type-%08d", len(a.results)+1)
			a.results[result] = resultToken
		}
	}
	a.pending = &objectPhaseEntry{rendered: rendered, queried: queried, result: result, witness: witness, expectedQuery: a.recorder.queries + 1,
		receipt: objectPhaseReceipt{Sequence: a.count, Pass: pass.report.Token, Program: pass.report.Program, Checker: pass.report.Checker, Phase: pass.report.Name + "_types", Witness: witnessToken, Result: resultToken, Rendered: phaseOracleDescribe(rendered), Queried: phaseOracleDescribe(queried), Candidate: objectPhaseIsCandidate(queried)}}
	return result
}

func (a *objectPhaseAdapter) returned(c *checker.Checker, rendered, queried *ast.Node, result *checker.Type) {
	entry, pass := a.pending, a.active
	if entry == nil || pass == nil || c != pass.program.checker || entry.rendered != rendered || entry.queried != queried || entry.result != result || entry.expectedQuery != a.recorder.queries {
		panic("object witness return does not match the actual walker query")
	}
	entry.receipt.Query = fmt.Sprintf("query-%08d", a.recorder.queries)
	entry.receipt.Operation = fmt.Sprintf("operation-%08d", a.recorder.sequence)
	data, err := json.Marshal(entry.receipt)
	if err != nil {
		panic(err)
	}
	data = append(data, '\n')
	if _, err := a.writer.Write(data); err != nil {
		panic(err)
	}
	a.bytes += len(data)
	pass.entries = append(pass.entries, entry)
	pass.report.Queries++
	if entry.receipt.Candidate {
		pass.report.Candidates++
	}
	a.pending = nil
}

func objectPhaseEndPass(p *phaseOracleProgram, recorder *phaseOracleRecorder, walk *phaseOracleWalk) {
	a := objectPhaseActive
	if a == nil {
		return
	}
	pass := a.active
	if pass == nil || pass.program != p || pass.report.Name != walk.Name || a.pending != nil || recorder != a.recorder {
		panic("object pass cannot close at this boundary")
	}
	if len(pass.entries) != len(walk.observation.typeQueries) {
		panic("object wrapper changed the direct query count")
	}
	for index, actual := range walk.observation.typeQueries {
		entry := pass.entries[index]
		if actual.renderedNode != entry.rendered || actual.queriedNode != entry.queried || actual.typ != entry.result {
			panic("object wrapper changed a recorded query return")
		}
	}
	if a.boundaryCheck != nil {
		a.boundaryCheck(a)
	}
	c, done := p.program.GetTypeCheckerForFileExclusive(context.Background(), p.program.SourceFiles()[0])
	defer done()
	if c != p.checker {
		panic("object pass closure checker changed")
	}
	if err := c.ProjectOracleEndObjectPass(pass.provider); err != nil {
		panic(err)
	}
	pass.report.Closed, pass.report.RawReturnsMatch = true, true
	pass.report.ClosedAfterSnapshot = walk.AfterSymbols
	pass.report.EndOperation = recorder.event(phaseOracleEvent{Program: pass.report.Program, Checker: pass.report.Checker, Phase: walk.Name, Operation: "object_provider_pass_closed"})
	a.active = nil
	objectPhaseQueryOverride = nil
	objectPhaseQueryReturnObserver = nil
}

func objectPhaseCounters(c *checker.Checker) objectPhaseAllocations {
	return objectPhaseAllocations{Types: c.TypeCount, Symbols: c.SymbolCount, Signatures: c.SignatureCount}
}

func (a *objectPhaseAdapter) verify() error {
	if a.active != nil || a.pending != nil || len(a.passes) != 2 || !a.passes[0].report.Closed || !a.passes[1].report.Closed {
		return fmt.Errorf("both artifact passes must close before verification")
	}
	if a.report.VerificationBegin != "" {
		return fmt.Errorf("object verification cannot run twice")
	}
	cold, warm := a.passes[0], a.passes[1]
	if cold.program != warm.program || cold.report.Name != "first" || warm.report.Name != "second" || len(cold.entries) != len(warm.entries) {
		a.report.PairingState = "mismatch"
		return fmt.Errorf("object query pass order or counts differ")
	}
	for index, left := range cold.entries {
		right := warm.entries[index]
		if left.rendered != right.rendered || left.queried != right.queried || left.receipt.Candidate != right.receipt.Candidate {
			a.report.PairingState = "mismatch"
			return fmt.Errorf("object query slots differ")
		}
	}
	a.report.PairingState = "exact_query_slots_and_nodes"
	p := warm.program
	c, done := p.program.GetTypeCheckerForFileExclusive(context.Background(), p.program.SourceFiles()[0])
	defer done()
	if c != p.checker {
		return fmt.Errorf("object verifier checker changed")
	}
	queries := a.recorder.queries
	a.report.VerificationBegin = a.recorder.event(phaseOracleEvent{Program: p.report.Token, Checker: p.report.Checker, Phase: "after_all_artifacts", Operation: "object_verification_phase_begin"})
	a.report.AllocationsBefore = objectPhaseCounters(c)
	for index, left := range cold.entries {
		right := warm.entries[index]
		if !left.receipt.Candidate {
			continue
		}
		pair := objectPhasePair{ColdQuery: left.receipt.Query, WarmQuery: right.receipt.Query, ColdWitness: left.receipt.Witness, WarmWitness: right.receipt.Witness, ColdResult: left.receipt.Result, WarmResult: right.receipt.Result, Queried: left.receipt.Queried, PointerEqual: left.result == right.result}
		pair.BeginOperation = a.recorder.event(phaseOracleEvent{Program: p.report.Token, Checker: p.report.Checker, Phase: "after_all_artifacts", Operation: "object_pair_verification_begin"})
		pair.Verification = c.ProjectOracleVerifyObjectsAfterArtifacts(left.witness, right.witness)
		pair.EndOperation = a.recorder.event(phaseOracleEvent{Program: p.report.Token, Checker: p.report.Checker, Phase: "after_all_artifacts", Operation: "object_pair_verification_end"})
		a.report.Pairs = append(a.report.Pairs, pair)
		a.report.VerificationStates[pair.Verification.State]++
		if pair.Verification.Relation.Called {
			a.report.RelationCalls++
		}
		a.report.IdentityCacheDelta += pair.Verification.Relation.IdentityCacheDelta
		a.report.DiagnosticCodes = append(a.report.DiagnosticCodes, pair.Verification.Relation.DiagnosticCodes...)
		a.report.SuggestionCodes = append(a.report.SuggestionCodes, pair.Verification.Relation.SuggestionCodes...)
	}
	a.report.AllocationsAfter = objectPhaseCounters(c)
	a.report.VerificationEnd = a.recorder.event(phaseOracleEvent{Program: p.report.Token, Checker: p.report.Checker, Phase: "after_all_artifacts", Operation: "object_verification_phase_end"})
	if a.recorder.queries != queries {
		return fmt.Errorf("verification added artifact queries")
	}
	a.report.State = "proof_results_recorded_no_acceptance_change"
	return nil
}

func (a *objectPhaseAdapter) closeReceipts() {
	if err := a.writer.Flush(); err != nil {
		panic(err)
	}
	if err := a.file.Close(); err != nil {
		panic(err)
	}
	a.report.Receipts = &phaseOracleTrace{Path: "object-query-receipts.jsonl", SHA256: hex.EncodeToString(a.hasher.Sum(nil)), Bytes: a.bytes, Events: a.count, Queries: a.count, TokenScope: "actual_artifact_query_returns_bound_to_opaque_provider_handles"}
}

func objectPhaseCheckArtifacts(output string, phase *phaseOracleReport) bool {
	check := func(artifact projectOracleArtifact) bool {
		if artifact.State == "no_content" || artifact.State == "unavailable" {
			return true
		}
		if artifact.Path == nil || artifact.SHA256 == nil || artifact.Bytes == nil {
			return false
		}
		data, err := os.ReadFile(filepath.Join(output, *artifact.Path))
		return err == nil && len(data) == *artifact.Bytes && projectOracleHash(data) == *artifact.SHA256
	}
	for _, program := range phase.Programs {
		for _, snapshot := range program.Snapshots {
			if !check(snapshot.Errors) {
				return false
			}
		}
		for _, walk := range program.ArtifactWalks {
			if !check(walk.Types) || !check(walk.Symbols) {
				return false
			}
		}
	}
	return true
}

func runObjectPhase(t *testing.T, config, output string, enabled bool, provenance json.RawMessage, boundaryCheck func(*objectPhaseAdapter)) (phase *phaseOracleReport, proof *objectPhaseProof) {
	t.Helper()
	if err := os.Mkdir(output, 0o700); err != nil {
		t.Fatal(err)
	}
	phase = newPhaseOracleReport(config)
	phase.Provenance = provenance
	phase.RunID = os.Getenv("TS_OBJECT_PHASE_RUN_ID")
	started := time.Now()
	recorder := newPhaseOracleRecorder(output)
	var adapter *objectPhaseAdapter
	if enabled {
		adapter = newObjectPhaseAdapter(output, config, recorder, provenance)
		adapter.boundaryCheck = boundaryCheck
		objectPhaseActive = adapter
		proof = adapter.report
	}
	defer func() {
		if failure := recover(); failure != nil {
			phase.Outcome = "oracle_error"
			phase.Failure = projectOracleString(fmt.Sprint(failure))
			if proof != nil {
				proof.State = "adapter_error"
				proof.Reason = fmt.Sprint(failure)
			}
			t.Error(failure)
		}
		objectPhaseActive = nil
		objectPhaseQueryOverride = nil
		objectPhaseQueryReturnObserver = nil
		projectOracleActiveObserver = nil
		phaseOracleActiveQueryObserver = nil
		if t.Failed() && phase.Outcome != "oracle_error" {
			phase.Outcome = "invariant_error"
			phase.Failure = projectOracleString("the pinned baseline renderer rejected an invariant")
		}
		if adapter != nil {
			adapter.closeReceipts()
			if err := projectOracleWriteJSON(filepath.Join(output, "object-witnesses.json"), proof); err != nil {
				t.Error(err)
			}
		}
		phase.Trace = recorder.close()
		phase.CheckerBindings = recorder.bindings
		phase.Runtime = projectOracleRuntimeSnapshot(started)
		if err := projectOracleWriteJSON(filepath.Join(output, "report.json"), phase); err != nil {
			t.Error(err)
		}
	}()
	phaseOracleRun(t, config, output, phase, recorder)
	if t.Failed() && phase.Outcome != "oracle_error" {
		phase.Outcome = "invariant_error"
		phase.Failure = projectOracleString("the pinned baseline renderer rejected an invariant")
	}
	if adapter == nil {
		return
	}
	proof.RawChecks, proof.Outcome = phase.PointerChecks, phase.Outcome
	for _, program := range phase.Programs {
		if program.GraphSHA256 != nil {
			proof.InputGraphHashes[program.Token] = *program.GraphSHA256
		}
	}
	before, err := json.Marshal(phase)
	if err != nil {
		panic(err)
	}
	proof.PrimaryBefore = projectOracleHash(before)
	if err := adapter.verify(); err != nil {
		proof.State = "incomplete"
		proof.Reason = err.Error()
	}
	after, err := json.Marshal(phase)
	if err != nil {
		panic(err)
	}
	proof.PrimaryAfter = projectOracleHash(after)
	proof.PrimaryUnchanged = proof.PrimaryBefore == proof.PrimaryAfter
	proof.ArtifactsUnchanged = objectPhaseCheckArtifacts(output, phase)
	if !proof.PrimaryUnchanged || !proof.ArtifactsUnchanged {
		proof.State = "adapter_error"
		proof.Reason = "object verification changed frozen phase evidence"
		t.Error(proof.Reason)
	}
	return
}

func TestObjectWitnessPhaseProject(t *testing.T) {
	config, output := os.Getenv("TS_OBJECT_PHASE_CONFIG"), os.Getenv("TS_OBJECT_PHASE_OUTPUT")
	if config == "" && output == "" {
		t.Skip("object phase environment is not set")
	}
	provenance, err := os.ReadFile(os.Getenv("TS_OBJECT_PHASE_PROVENANCE"))
	if err != nil {
		t.Fatal(err)
	}
	if err := projectOracleValidateProvenance(provenance); err != nil {
		t.Fatal(err)
	}
	phase, _ := runObjectPhase(t, config, output, true, provenance, nil)
	if phase.Outcome != "complete" {
		t.Errorf("preserved phase outcome: %s", phase.Outcome)
	}
}
