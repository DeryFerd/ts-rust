package tsbaseline

import (
	"github.com/microsoft/typescript-go/internal/ast"
	"github.com/microsoft/typescript-go/internal/checker"
)

var objectPhaseQueryOverride func(*checker.Checker, *ast.Node, *ast.Node) *checker.Type
var objectPhaseQueryReturnObserver func(*checker.Checker, *ast.Node, *ast.Node, *checker.Type)

func objectPhaseTypeQuery(c *checker.Checker, rendered, queried *ast.Node) *checker.Type {
	if objectPhaseQueryOverride != nil {
		return objectPhaseQueryOverride(c, rendered, queried)
	}
	return c.GetTypeAtLocation(queried)
}

func objectPhaseTypeQueryReturned(c *checker.Checker, rendered, queried *ast.Node, result *checker.Type) {
	if objectPhaseQueryReturnObserver != nil {
		objectPhaseQueryReturnObserver(c, rendered, queried, result)
	}
}
