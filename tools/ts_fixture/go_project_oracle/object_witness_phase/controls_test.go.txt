func objectPhaseReadLines[T any](t *testing.T, file string) []T {
	t.Helper()
	input, err := os.Open(file)
	if err != nil {
		t.Fatal(err)
	}
	defer input.Close()
	result := []T{}
	scanner := bufio.NewScanner(input)
	for scanner.Scan() {
		var item T
		if err := json.Unmarshal(scanner.Bytes(), &item); err != nil {
			t.Fatal(err)
		}
		result = append(result, item)
	}
	if err := scanner.Err(); err != nil {
		t.Fatal(err)
	}
	return result
}

func objectPhaseAssertReports(t *testing.T, plain, adapted *phaseOracleReport, proof *objectPhaseProof, plainOut, adaptedOut string) {
	t.Helper()
	if proof.State != "proof_results_recorded_no_acceptance_change" || proof.PairingState != "exact_query_slots_and_nodes" {
		t.Fatalf("adapter did not complete: %+v", proof)
	}
	if plain.Outcome != adapted.Outcome || plain.PointerChecks != adapted.PointerChecks || proof.RawChecks != adapted.PointerChecks || proof.Outcome != adapted.Outcome {
		t.Fatal("object evidence changed an existing phase gate")
	}
	if adapted.Outcome == "complete" || adapted.FreshDiagnostics.State != "unavailable" || adapted.FreshDiagnostics.Equal != nil || proof.FreshDiagnosticEquality != nil || proof.ParityClaimed {
		t.Fatal("bounded object evidence became project parity")
	}
	if !proof.PrimaryUnchanged || proof.PrimaryBefore != proof.PrimaryAfter || !proof.ArtifactsUnchanged {
		t.Fatal("verification changed frozen evidence")
	}
	if proof.ProviderCommit != objectPhaseProviderCommit || proof.SourceCommit == "" || proof.BinarySHA256 == "" || proof.ConfigSHA256 == "" {
		t.Fatal("proof provenance is missing")
	}
	if len(proof.Passes) != 2 || len(adapted.Programs) != 2 || adapted.Programs[0].ArtifactQueries != 0 {
		t.Fatal("independent source-only Program was lost")
	}
	var typeQueries int
	for programIndex, program := range adapted.Programs {
		old := plain.Programs[programIndex]
		if program.Role != old.Role || program.EligibleProjectSources != old.EligibleProjectSources || len(program.Snapshots) != len(old.Snapshots) {
			t.Fatal("program structure changed")
		}
		if !reflect.DeepEqual(program.Graph, old.Graph) {
			t.Fatal("phase input graph changed")
		}
		for index, snapshot := range program.Snapshots {
			previous := old.Snapshots[index]
			if snapshot.Phase != previous.Phase || !reflect.DeepEqual(snapshot.Diagnostics, previous.Diagnostics) || !reflect.DeepEqual(snapshot.Errors, previous.Errors) || !reflect.DeepEqual(snapshot.ErrorsFileOrder, previous.ErrorsFileOrder) {
				t.Fatalf("diagnostic phase changed: %s", snapshot.Phase)
			}
		}
		for index, walk := range program.ArtifactWalks {
			previous := old.ArtifactWalks[index]
			if walk.HadErrorBaseline != previous.HadErrorBaseline || walk.Counts != previous.Counts || !reflect.DeepEqual(walk.Types, previous.Types) || !reflect.DeepEqual(walk.Symbols, previous.Symbols) {
				t.Fatal("artifact observations changed")
			}
			typeQueries += walk.Counts.TypeQueries
		}
	}
	events := objectPhaseReadLines[phaseOracleEvent](t, filepath.Join(adaptedOut, "sequence.jsonl"))
	oldEvents := objectPhaseReadLines[phaseOracleEvent](t, filepath.Join(plainOut, "sequence.jsonl"))
	queryProjection := func(events []phaseOracleEvent) []phaseOracleEvent {
		queries := []phaseOracleEvent{}
		for _, event := range events {
			if event.Query == "" {
				continue
			}
			event.Sequence, event.Token = 0, ""
			queries = append(queries, event)
		}
		return queries
	}
	if !reflect.DeepEqual(queryProjection(events), queryProjection(oldEvents)) {
		t.Fatal("actual query order, arguments, or counts changed")
	}
	byOperation := map[string]phaseOracleEvent{}
	for _, event := range events {
		byOperation[event.Token] = event
	}
	var lastClose uint64
	for _, pass := range proof.Passes {
		if !pass.Closed || !pass.RawReturnsMatch || pass.Program != adapted.Programs[1].Token || pass.Checker != adapted.Programs[1].Checker {
			t.Fatal("provider pass did not use the query Program and actual returns")
		}
		closed := byOperation[pass.EndOperation]
		if closed.Operation != "object_provider_pass_closed" {
			t.Fatal("provider pass closure is not recorded")
		}
		lastClose = max(lastClose, closed.Sequence)
		var snapshot *phaseOracleSnapshot
		for _, item := range adapted.Programs[1].Snapshots {
			if item.Token == pass.ClosedAfterSnapshot {
				copy := item
				snapshot = &copy
			}
		}
		if snapshot == nil || !strings.HasSuffix(snapshot.Phase, "after_symbols") || byOperation[snapshot.EndOperation].Sequence >= closed.Sequence {
			t.Fatal("provider closed before artifact observations ended")
		}
	}
	if byOperation[proof.VerificationBegin].Sequence <= lastClose {
		t.Fatal("verification ran before both passes closed")
	}
	var allocations objectPhaseAllocations
	for _, pair := range proof.Pairs {
		if byOperation[pair.BeginOperation].Sequence <= lastClose || byOperation[pair.EndOperation].Sequence <= byOperation[pair.BeginOperation].Sequence {
			t.Fatal("pair verification ordering changed")
		}
		allocations.Types += pair.Verification.Relation.TypeAllocations
		allocations.Symbols += pair.Verification.Relation.SymbolAllocations
		allocations.Signatures += pair.Verification.Relation.SignatureAllocations
	}
	if allocations.Types != proof.AllocationsAfter.Types-proof.AllocationsBefore.Types || allocations.Symbols != proof.AllocationsAfter.Symbols-proof.AllocationsBefore.Symbols || allocations.Signatures != proof.AllocationsAfter.Signatures-proof.AllocationsBefore.Signatures {
		t.Fatal("relation allocation side effects were not retained")
	}
	receipts := objectPhaseReadLines[objectPhaseReceipt](t, filepath.Join(adaptedOut, proof.Receipts.Path))
	if len(receipts) != typeQueries || proof.Receipts.Queries != uint64(typeQueries) {
		t.Fatal("a direct type query has no exact provider receipt")
	}
	seen := map[string]bool{}
	for index, receipt := range receipts {
		if receipt.Sequence != uint64(index+1) || receipt.Witness != fmt.Sprintf("object-witness-%08d", index+1) || seen[receipt.Witness] {
			t.Fatal("provider witness handles were not recorded uniquely")
		}
		seen[receipt.Witness] = true
		query := byOperation[receipt.Operation]
		if query.Query != receipt.Query || query.Operation != "get_type_at_location" || query.Program != receipt.Program || query.Checker != receipt.Checker || !reflect.DeepEqual(query.Rendered, receipt.Rendered) || !reflect.DeepEqual(query.Queried, receipt.Queried) {
			t.Fatal("provider receipt does not name its actual query")
		}
	}
	data, err := os.ReadFile(filepath.Join(adaptedOut, proof.Receipts.Path))
	if err != nil {
		t.Fatal(err)
	}
	if projectOracleHash(data) != proof.Receipts.SHA256 || len(data) != proof.Receipts.Bytes {
		t.Fatal("receipt checksum differs")
	}
}

func TestObjectPhaseControls(t *testing.T) {
	root := os.Getenv("TS_OBJECT_PHASE_OUTPUT")
	if root == "" {
		t.Skip("object phase control output is not set")
	}
	if err := os.Mkdir(root, 0o700); err != nil {
		t.Fatal(err)
	}
	provenance, err := os.ReadFile(os.Getenv("TS_OBJECT_PHASE_PROVENANCE"))
	if err != nil {
		t.Fatal(err)
	}
	if err := projectOracleValidateProvenance(provenance); err != nil {
		t.Fatal(err)
	}
	cases := []struct {
		name, source string
		verified     bool
	}{
		{"empty", `export const object = {};`, true},
		{"nested_parentheses", `export const object = ({child: ({value: 1}), truth: true});`, true},
		{"function_cache", `function stable(value: number): number { return value; } export const object = {stable};`, true},
		{"computed_unproved", `export const object = {["value"]: 1};`, false},
		{"const_unproved", `export const object = {value: 1} as const;`, false},
		{"existing_source_error", `export const value: number = "bad";`, false},
		{"class_base_fallback", `declare const Base: any; export class Derived extends Base {}`, false},
		{"query_added_diagnostics", `export namespace JSX { export interface IntrinsicElements { div?: string; } }
export interface IntrinsicElements extends JSX.IntrinsicElements {}
export function makeCache(): () => Promise<number> {
  return async function cache() { const cache = 1; return cache; };
}
export function makeEtag(): () => Promise<string | null> {
  return async function etag() { let etag: string | null = null; return etag; };
}`, false},
	}
	for _, test := range cases {
		t.Run(test.name, func(t *testing.T) {
			directory := filepath.Join(root, test.name)
			input := filepath.Join(directory, "input")
			if err := os.MkdirAll(input, 0o700); err != nil {
				t.Fatal(err)
			}
			config := filepath.Join(input, "tsconfig.json")
			for file, contents := range map[string]string{"input.ts": test.source, "tsconfig.json": `{"compilerOptions":{"strict":true,"declaration":true,"skipLibCheck":true,"noUnusedLocals":true,"noUnusedParameters":true,"target":"ES2022","module":"ES2020","moduleResolution":"Bundler","types":[]},"files":["input.ts"]}`} {
				if err := os.WriteFile(filepath.Join(input, file), []byte(contents), 0o600); err != nil {
					t.Fatal(err)
				}
			}
			plainOut, adaptedOut := filepath.Join(directory, "plain"), filepath.Join(directory, "adapted")
			plain, _ := runObjectPhase(t, config, plainOut, false, provenance, nil)
			guards := 0
			adapted, proof := runObjectPhase(t, config, adaptedOut, true, provenance, func(adapter *objectPhaseAdapter) {
				guards++
				if err := adapter.verify(); err == nil || adapter.report.RelationCalls != 0 || len(adapter.report.Pairs) != 0 {
					t.Fatal("open artifact pass allowed verification")
				}
			})
			if guards != 4 {
				t.Fatal("open-pass guards did not run at both boundaries")
			}
			objectPhaseAssertReports(t, plain, adapted, proof, plainOut, adaptedOut)
			if test.verified && proof.VerificationStates["verified"] == 0 {
				t.Fatalf("ordinary objects have no verified provider result: %+v", proof.Pairs)
			}
			if test.name == "empty" {
				if len(proof.Pairs) != 1 || proof.Pairs[0].PointerEqual || !proof.Pairs[0].Verification.Relation.Called || proof.Pairs[0].Verification.Relation.Identical == nil || !*proof.Pairs[0].Verification.Relation.Identical || adapted.Outcome != "invariant_error" {
					t.Fatal("verified fresh objects bypassed the raw pointer gate")
				}
			}
			if test.name == "computed_unproved" || test.name == "const_unproved" {
				if len(proof.Pairs) == 0 || proof.VerificationStates["verified"] != 0 || proof.RelationCalls != 0 {
					t.Fatal("unsupported object context gained relation acceptance")
				}
			}
			if test.name == "class_base_fallback" && adapted.Programs[1].ArtifactWalks[0].Counts.TypeQueries <= adapted.Programs[1].ArtifactWalks[0].Counts.RenderedTypes {
				t.Fatal("class-base fallback did not exercise both actual calls")
			}
			if test.name == "query_added_diagnostics" {
				for _, snapshot := range adapted.Programs[0].Snapshots {
					if len(snapshot.Diagnostics) != 0 {
						t.Fatal("source-only diagnostics were changed")
					}
				}
				for _, walk := range adapted.Programs[1].ArtifactWalks {
					if walk.HadErrorBaseline {
						t.Fatal("query-added diagnostics changed the renderer input")
					}
				}
				if !slices.ContainsFunc(adapted.Programs[1].Snapshots, func(snapshot phaseOracleSnapshot) bool {
					return snapshot.Phase == "first_after_types" && len(snapshot.Diagnostics) == 5
				}) {
					t.Fatal("query-added diagnostic phase was lost")
				}
			}
			if test.name == "existing_source_error" {
				for _, walk := range adapted.Programs[1].ArtifactWalks {
					if !walk.HadErrorBaseline {
						t.Fatal("initial error renderer input was lost")
					}
				}
			}
			t.Logf("%s: state=%s raw_type_changes=%d provider=%v relations=%d", test.name, adapted.Outcome, adapted.PointerChecks.ChangedTypeQueries, proof.VerificationStates, proof.RelationCalls)
		})
	}
}
