const projectOracleUpstreamSHA = "dc37b5249ab60e2bbce936f71b883e6c8136167e"

type projectOracleArtifact struct {
	State  string  `json:"state"`
	Path   *string `json:"path"`
	SHA256 *string `json:"sha256"`
	Bytes  *int    `json:"bytes"`
	Reason *string `json:"reason"`
}

type projectOracleDiagnostic struct {
	File        *string                   `json:"file"`
	Start       int                       `json:"start"`
	Length      int                       `json:"length"`
	Code        int32                     `json:"code"`
	Category    string                    `json:"category"`
	Message     string                    `json:"message"`
	Unnecessary bool                      `json:"unnecessary"`
	Deprecated  bool                      `json:"deprecated"`
	Related     []projectOracleDiagnostic `json:"related"`
}

type projectOracleWalk struct {
	TypeASTVisits   int `json:"type_ast_visits"`
	SymbolASTVisits int `json:"symbol_ast_visits"`
	TypeQueries     int `json:"type_queries"`
	SymbolQueries   int `json:"symbol_queries"`
	SymbolsReturned int `json:"symbols_returned"`
}

type projectOraclePass struct {
	Diagnostics []projectOracleDiagnostic `json:"diagnostics"`
	ErrorsFileOrder []string               `json:"errors_file_order"`
	Errors      projectOracleArtifact     `json:"errors"`
	Types       projectOracleArtifact     `json:"types"`
	Symbols     projectOracleArtifact     `json:"symbols"`
	Walk        projectOracleWalk         `json:"walk"`
}

type projectOracleReset struct {
	File          string `json:"file"`
	TypeChecked   bool   `json:"type_checked"`
	UnusedChecked bool   `json:"unused_checked"`
}

type projectOracleReplay struct {
	State            string               `json:"state"`
	Sources          []projectOracleReset `json:"sources"`
	DiagnosticsEqual bool                 `json:"diagnostics_equal"`
	ArtifactsEqual   bool                 `json:"artifacts_equal"`
	WalkEqual        bool                 `json:"walk_equal"`
	TypeIDsEqual     bool                 `json:"type_identities_equal"`
	SymbolIDsEqual   bool                 `json:"symbol_identities_equal"`
}

type projectOracleRuntime struct {
	GoVersion        string            `json:"go_version"`
	OS               string            `json:"os"`
	Architecture     string            `json:"architecture"`
	CPUCount         int               `json:"cpu_count"`
	StartedAt        string            `json:"started_at"`
	ElapsedNanos     int64             `json:"elapsed_nanos"`
	ExecutableSHA256 *string           `json:"executable_sha256"`
	Arguments        []string          `json:"arguments"`
	Environment      map[string]string `json:"environment"`
	ProcessStatus    *string           `json:"process_status"`
	ProcessLimits    *string           `json:"process_limits"`
	CgroupMemoryMax  *string           `json:"cgroup_memory_max"`
	BuildInfo        *string           `json:"build_info"`
}

type projectOracleReport struct {
	SchemaVersion  int                  `json:"schema_version"`
	Implementation string               `json:"implementation"`
	UpstreamSHA    string               `json:"upstream_sha"`
	RunID          string               `json:"run_id"`
	Header         string               `json:"header"`
	Outcome        string               `json:"outcome"`
	Failure        *string              `json:"failure"`
	Graph          *projectOracleGraph  `json:"graph"`
	GraphSHA256    *string              `json:"graph_sha256"`
	Cold           projectOraclePass    `json:"cold"`
	Warm           projectOraclePass    `json:"warm"`
	Replay         projectOracleReplay  `json:"replay"`
	Provenance     json.RawMessage      `json:"provenance"`
	Runtime        projectOracleRuntime `json:"runtime"`
}

type projectOracleTypeIdentity struct {
	node *ast.Node
	typ  *checker.Type
}

type projectOracleSymbolIdentity struct {
	node   *ast.Node
	symbol *ast.Symbol
}

type projectOracleObservation struct {
	walk         projectOracleWalk
	typeVisits   []*ast.Node
	symbolVisits []*ast.Node
	types        []projectOracleTypeIdentity
	symbols      []projectOracleSymbolIdentity
	texts        [3]string
}

func (observation *projectOracleObservation) observer() *projectOracleObserver {
	return &projectOracleObserver{
		visit: func(node *ast.Node, symbolWalk bool) {
			if symbolWalk {
				observation.walk.SymbolASTVisits++
				observation.symbolVisits = append(observation.symbolVisits, node)
			} else {
				observation.walk.TypeASTVisits++
				observation.typeVisits = append(observation.typeVisits, node)
			}
		},
		typ: func(node *ast.Node, typ *checker.Type) {
			observation.walk.TypeQueries++
			observation.types = append(observation.types, projectOracleTypeIdentity{node, typ})
		},
		symbol: func(node *ast.Node, symbol *ast.Symbol) {
			observation.walk.SymbolQueries++
			if symbol != nil {
				observation.walk.SymbolsReturned++
			}
			observation.symbols = append(observation.symbols, projectOracleSymbolIdentity{node, symbol})
		},
	}
}

func TestProjectOracle(t *testing.T) {
	configPath := os.Getenv("TS_RUST_ORACLE_PROJECT")
	output := os.Getenv("TS_RUST_ORACLE_OUTPUT")
	if configPath == "" && output == "" {
		t.Skip("project oracle environment is not set")
	}
	if !filepath.IsAbs(configPath) || !filepath.IsAbs(output) {
		t.Fatal("project config and output paths must be absolute")
	}
	if err := os.Mkdir(output, 0o700); err != nil {
		t.Fatal(err)
	}
	header := os.Getenv("TS_RUST_ORACLE_HEADER")
	if header == "" {
		header = "project"
	}
	started := time.Now()
	report := projectOracleNewReport(header)
	report.RunID = os.Getenv("TS_RUST_ORACLE_RUN_ID")
	defer func() {
		if value := recover(); value != nil {
			report.Outcome = "oracle_error"
			report.Failure = projectOracleString(fmt.Sprintf("panic: %v\n%s", value, debug.Stack()))
		}
		projectOracleActiveObserver = nil
		report.Runtime = projectOracleRuntimeSnapshot(started)
		if t.Failed() && report.Outcome == "complete" {
			report.Outcome = "invariant_error"
			report.Failure = projectOracleString("the pinned baseline renderer rejected an invariant")
		}
		if err := projectOracleWriteJSON(filepath.Join(output, "report.json"), report); err != nil {
			t.Error(err)
		}
		if report.Outcome != "complete" {
			t.Errorf("project oracle outcome: %s", report.Outcome)
		}
	}()
	provenancePath := os.Getenv("TS_RUST_ORACLE_PROVENANCE")
	provenance, err := os.ReadFile(provenancePath)
	if err != nil || !json.Valid(provenance) {
		report.Failure = projectOracleString(fmt.Sprintf("cannot read valid build provenance from %q: %v", provenancePath, err))
		return
	}
	report.Provenance = provenance
	if err := projectOracleValidateProvenance(provenance); err != nil {
		report.Failure = projectOracleString(err.Error())
		return
	}
	projectOracleRun(t, configPath, output, report)
}

func projectOracleValidateProvenance(data []byte) error {
	var manifest struct {
		State string `json:"state"`
		Upstream struct {
			SHA string `json:"sha"`
			Dirty string `json:"dirty"`
		} `json:"upstream"`
		Executable *struct {
			SHA256 string `json:"sha256"`
		} `json:"executable"`
	}
	if err := json.Unmarshal(data, &manifest); err != nil {
		return err
	}
	if manifest.State != "built" || manifest.Upstream.SHA != projectOracleUpstreamSHA || manifest.Upstream.Dirty != "" || manifest.Executable == nil {
		return fmt.Errorf("build provenance does not describe a successful pinned oracle build")
	}
	executable, err := os.Executable()
	if err != nil { return err }
	binary, err := os.ReadFile(executable)
	if err != nil { return err }
	if projectOracleHash(binary) != manifest.Executable.SHA256 {
		return fmt.Errorf("oracle executable does not match build provenance")
	}
	return nil
}

func projectOracleNewReport(header string) *projectOracleReport {
	return &projectOracleReport{
		SchemaVersion: 1, Implementation: "typescript-go", UpstreamSHA: projectOracleUpstreamSHA,
		Header: header, Outcome: "oracle_error", Provenance: json.RawMessage("null"),
		Cold: projectOracleUnavailablePass("cold run did not complete"),
		Warm: projectOracleUnavailablePass("forced replay did not complete"),
		Replay: projectOracleReplay{State: "unavailable", Sources: []projectOracleReset{}},
	}
}

func projectOracleRun(t *testing.T, configPath, output string, report *projectOracleReport) {
	if !bundled.Embedded {
		report.Failure = projectOracleString("the oracle requires the pinned embedded libraries")
		return
	}
	fs := newProjectOracleFS(bundled.WrapFS(osvfs.FS()))
	host := compiler.NewCompilerHost(filepath.ToSlash(filepath.Dir(configPath)), fs, bundled.LibPath(), nil, nil)
	config, loadDiagnostics := tsoptions.GetParsedCommandLineOfConfigFile(filepath.ToSlash(configPath), nil, nil, host, nil)
	if config == nil {
		report.Outcome = "config_error"
		report.Failure = projectOracleString("the normal compiler host could not load the config")
		report.Cold.Diagnostics = projectOracleDiagnostics(loadDiagnostics)
		report.Cold.Errors = projectOracleWriteArtifact(output, "cold.errors.txt", projectOracleErrorBaseline(t, nil, loadDiagnostics, &report.Cold))
		report.Graph = projectOracleUnloadedGraph(configPath, fs)
		projectOracleHashGraph(report)
		return
	}
	fs.setPhase("program")
	program := compiler.NewProgram(compiler.ProgramOptions{Host: host, Config: config, SingleThreaded: core.TSTrue})
	files := projectOracleArtifactFiles(program)
	defer func() {
		report.Graph = projectOracleSnapshotGraph(program, fs, files)
		projectOracleHashGraph(report)
	}()
	diagnostics := projectOracleCollectDiagnostics(program, loadDiagnostics)
	if config.CompilerOptions().NoCheck.IsTrue() || len(files) == 0 {
		report.Cold.Diagnostics = projectOracleDiagnostics(diagnostics)
		report.Cold.Errors = projectOracleWriteArtifact(output, "cold.errors.txt", projectOracleErrorBaseline(t, files, diagnostics, &report.Cold))
		if config.CompilerOptions().NoCheck.IsTrue() {
			report.Outcome = "no_check"
			report.Failure = projectOracleString("the project disables semantic checking")
		} else {
			report.Outcome = "no_sources"
			report.Failure = projectOracleString("the project has no non-default-library sources")
		}
		return
	}
	fs.setPhase("cold_artifacts")
	cold := projectOracleCapture(t, program, files, diagnostics, output, "cold", report.Header, &report.Cold)
	fs.setPhase("replay")
	for _, file := range program.SourceFiles() {
		if program.SkipTypeChecking(file, false) {
			continue
		}
		fileChecker, done := program.GetTypeCheckerForFileExclusive(context.Background(), file)
		typeChecked, unusedChecked := fileChecker.ProjectOracleResetSource(file)
		done()
		report.Replay.Sources = append(report.Replay.Sources, projectOracleReset{projectOraclePath(file.FileName()), typeChecked, unusedChecked})
		if !typeChecked {
			report.Outcome = "invariant_error"
			report.Failure = projectOracleString("an eligible source was not checked before forced replay: " + file.FileName())
			return
		}
	}
	warmDiagnostics := projectOracleCollectDiagnostics(program, loadDiagnostics)
	fs.setPhase("warm_artifacts")
	warm := projectOracleCapture(t, program, files, warmDiagnostics, output, "warm", report.Header, &report.Warm)
	report.Replay.State = "complete"
	report.Replay.DiagnosticsEqual = reflect.DeepEqual(report.Cold.Diagnostics, report.Warm.Diagnostics)
	report.Replay.ArtifactsEqual = cold.texts == warm.texts
	report.Replay.WalkEqual = cold.walk == warm.walk && slices.Equal(cold.typeVisits, warm.typeVisits) && slices.Equal(cold.symbolVisits, warm.symbolVisits)
	report.Replay.TypeIDsEqual = slices.Equal(cold.types, warm.types)
	report.Replay.SymbolIDsEqual = slices.Equal(cold.symbols, warm.symbols)
	if !report.Replay.DiagnosticsEqual || !report.Replay.ArtifactsEqual || !report.Replay.WalkEqual || !report.Replay.TypeIDsEqual || !report.Replay.SymbolIDsEqual {
		report.Outcome = "invariant_error"
		report.Failure = projectOracleString("forced source replay changed diagnostics, artifacts, visits, or retained identities")
		return
	}
	if len(program.GetConfigFileParsingDiagnostics()) != 0 {
		report.Outcome = "config_error"
		report.Failure = projectOracleString("the normal config loader reported diagnostics")
		return
	}
	report.Outcome = "complete"
}

// Use Program diagnostics so comment directives and no-emit filtering stay active.
func projectOracleCollectDiagnostics(program *compiler.Program, loadDiagnostics []*ast.Diagnostic) []*ast.Diagnostic {
	ctx := context.Background()
	diagnostics := slices.Clone(loadDiagnostics)
	diagnostics = append(diagnostics, program.GetConfigFileParsingDiagnostics()...)
	diagnostics = append(diagnostics, program.GetProgramDiagnostics()...)
	diagnostics = append(diagnostics, program.GetSyntacticDiagnostics(ctx, nil)...)
	diagnostics = append(diagnostics, program.GetSemanticDiagnostics(ctx, nil)...)
	diagnostics = append(diagnostics, program.GetGlobalDiagnostics(ctx)...)
	if program.Options().GetEmitDeclarations() {
		diagnostics = append(diagnostics, program.GetDeclarationDiagnostics(ctx, nil)...)
	}
	return compiler.SortAndDeduplicateDiagnostics(diagnostics)
}

func projectOracleCapture(t *testing.T, program *compiler.Program, files []*harnessutil.TestFile, diagnostics []*ast.Diagnostic, output, prefix, header string, pass *projectOraclePass) *projectOracleObservation {
	observation := &projectOracleObservation{}
	defer func() { pass.Walk = observation.walk }()
	pass.Diagnostics = projectOracleDiagnostics(diagnostics)
	observation.texts[0] = projectOracleErrorBaseline(t, files, diagnostics, pass)
	pass.Errors = projectOracleWriteArtifact(output, prefix+".errors.txt", observation.texts[0])
	projectOracleActiveObserver = observation.observer()
	defer func() { projectOracleActiveObserver = nil }()
	walker := newTypeWriterWalker(program, len(diagnostics) != 0)
	observation.texts[1] = generateBaseline(files, walker, header, false)
	pass.Types = projectOracleWriteArtifact(output, prefix+".types", observation.texts[1])
	observation.texts[2] = generateBaseline(files, walker, header, true)
	pass.Symbols = projectOracleWriteArtifact(output, prefix+".symbols", observation.texts[2])
	return observation
}

func projectOracleErrorBaseline(t *testing.T, files []*harnessutil.TestFile, diagnostics []*ast.Diagnostic, pass *projectOraclePass) string {
	if len(diagnostics) == 0 {
		return baseline.NoContent
	}
	errorFiles := slices.Clone(files)
	seen := map[string]bool{}
	for _, file := range files {
		seen[file.UnitName] = true
	}
	extra := map[string]*ast.SourceFile{}
	for _, diagnostic := range diagnostics {
		if file := diagnostic.File(); file != nil && !seen[file.FileName()] && !isDefaultLibraryFile(file.FileName()) && !isBuiltFile(file.FileName()) {
			extra[file.FileName()] = file
		}
	}
	names := make([]string, 0, len(extra))
	for name := range extra { names = append(names, name) }
	slices.Sort(names)
	for _, name := range names {
		errorFiles = append(errorFiles, &harnessutil.TestFile{UnitName: name, Content: extra[name].Text()})
	}
	for _, file := range errorFiles {
		pass.ErrorsFileOrder = append(pass.ErrorsFileOrder, projectOraclePath(file.UnitName))
	}
	return GetErrorBaseline(t, errorFiles, diagnosticwriter.WrapASTDiagnostics(diagnostics), diagnosticwriter.CompareASTDiagnostics, false)
}

func projectOracleDiagnostics(diagnostics []*ast.Diagnostic) []projectOracleDiagnostic {
	result := make([]projectOracleDiagnostic, 0, len(diagnostics))
	for _, diagnostic := range diagnosticwriter.WrapASTDiagnostics(diagnostics) {
		result = append(result, projectOracleDiagnosticRecord(diagnostic))
	}
	return result
}

func projectOracleDiagnosticRecord(diagnostic diagnosticwriter.Diagnostic) projectOracleDiagnostic {
	var file *string
	if diagnostic.File() != nil {
		file = projectOracleString(projectOraclePath(diagnostic.File().FileName()))
	}
	result := projectOracleDiagnostic{
		File: file, Start: diagnostic.Pos(), Length: diagnostic.Len(), Code: diagnostic.Code(),
		Category: diagnostic.Category().Name(),
		Message: diagnosticwriter.FlattenDiagnosticMessage(diagnostic, "\n", locale.Default),
		Related: []projectOracleDiagnostic{},
	}
	if wrapped, ok := diagnostic.(*diagnosticwriter.ASTDiagnostic); ok {
		result.Unnecessary = wrapped.ReportsUnnecessary()
		result.Deprecated = wrapped.ReportsDeprecated()
	}
	for _, related := range diagnostic.RelatedInformation() {
		result.Related = append(result.Related, projectOracleDiagnosticRecord(related))
	}
	return result
}

func projectOracleWriteArtifact(output, name, content string) projectOracleArtifact {
	if content == baseline.NoContent {
		return projectOracleArtifact{State: "no_content", Reason: projectOracleString("the pinned renderer returned baseline.NoContent")}
	}
	if err := os.WriteFile(filepath.Join(output, name), []byte(content), 0o600); err != nil {
		panic(err)
	}
	size := len(content)
	return projectOracleArtifact{State: "present", Path: &name, SHA256: projectOracleString(projectOracleHash([]byte(content))), Bytes: &size}
}

func projectOracleUnavailablePass(reason string) projectOraclePass {
	absent := projectOracleArtifact{State: "unavailable", Reason: &reason}
	return projectOraclePass{Diagnostics: []projectOracleDiagnostic{}, ErrorsFileOrder: []string{}, Errors: absent, Types: absent, Symbols: absent}
}

func projectOracleHash(data []byte) string {
	digest := sha256.Sum256(data)
	return hex.EncodeToString(digest[:])
}

func projectOracleString(value string) *string { return &value }

func projectOracleWriteJSON(path string, value any) error {
	data, err := json.MarshalIndent(value, "", "  ")
	if err != nil {
		return err
	}
	return os.WriteFile(path, append(data, '\n'), 0o600)
}

func projectOracleHashGraph(report *projectOracleReport) {
	data, err := json.Marshal(report.Graph)
	if err != nil {
		panic(err)
	}
	report.GraphSHA256 = projectOracleString(projectOracleHash(data))
}

func projectOracleRuntimeSnapshot(started time.Time) projectOracleRuntime {
	result := projectOracleRuntime{
		GoVersion: runtime.Version(), OS: runtime.GOOS, Architecture: runtime.GOARCH,
		CPUCount: runtime.NumCPU(), StartedAt: started.UTC().Format(time.RFC3339Nano),
		ElapsedNanos: time.Since(started).Nanoseconds(), Arguments: slices.Clone(os.Args),
		Environment: map[string]string{},
		ProcessStatus: projectOracleReadOptional("/proc/self/status"),
		ProcessLimits: projectOracleReadOptional("/proc/self/limits"),
		CgroupMemoryMax: projectOracleReadOptional("/sys/fs/cgroup/memory.max"),
	}
	for _, key := range []string{"GOTOOLCHAIN", "GOPROXY", "GOCACHE", "GOMODCACHE", "GOFLAGS", "GOMAXPROCS", "GOMEMLIMIT", "TS_RUST_ORACLE_PROJECT", "TS_RUST_ORACLE_OUTPUT", "TS_RUST_ORACLE_HEADER", "TS_RUST_ORACLE_PROVENANCE", "TS_RUST_ORACLE_RUN_ID"} {
		if value, ok := os.LookupEnv(key); ok {
			result.Environment[key] = value
		}
	}
	if executable, err := os.Executable(); err == nil {
		if data, err := os.ReadFile(executable); err == nil {
			result.ExecutableSHA256 = projectOracleString(projectOracleHash(data))
		}
	}
	if info, ok := debug.ReadBuildInfo(); ok {
		result.BuildInfo = projectOracleString(info.String())
	}
	return result
}

func projectOracleReadOptional(path string) *string {
	data, err := os.ReadFile(path)
	if err != nil {
		return nil
	}
	return projectOracleString(string(data))
}

func TestProjectOracleMetadata(t *testing.T) {
	if projectOraclePath("bundled:///libs/lib.es5.d.ts") != "/__typescript/lib/lib.es5.d.ts" || projectOraclePath("/project/bundled/lib.es5.d.ts") != "/project/bundled/lib.es5.d.ts" {
		t.Fatal("only the exact bundled library prefix may change")
	}
	output := t.TempDir()
	absent := projectOracleWriteArtifact(output, "absent.errors.txt", baseline.NoContent)
	if absent.State != "no_content" || absent.Path != nil || absent.SHA256 != nil || absent.Bytes != nil {
		t.Fatal("absent artifacts must not contain paths or hashes")
	}
	content := "first\r\nsecond\r\n"
	present := projectOracleWriteArtifact(output, "cold.types", content)
	data, err := os.ReadFile(filepath.Join(output, "cold.types"))
	if err != nil || string(data) != content || present.SHA256 == nil || *present.SHA256 != projectOracleHash(data) {
		t.Fatal("artifact bytes or digest changed")
	}
	for _, invalid := range []string{`null`, `{}`, `{"state":"prepared"}`, `{"state":"built","upstream":{"sha":"wrong"}}`} {
		if projectOracleValidateProvenance([]byte(invalid)) == nil {
			t.Fatalf("accepted incomplete build provenance: %s", invalid)
		}
	}
}

func TestProjectOracleNoCheck(t *testing.T) {
	project := t.TempDir()
	if err := os.WriteFile(filepath.Join(project, "tsconfig.json"), []byte(`{"compilerOptions":{"noCheck":true},"files":["input.ts"]}`), 0o600); err != nil {
		t.Fatal(err)
	}
	if err := os.WriteFile(filepath.Join(project, "input.ts"), []byte("const input = 1;\n"), 0o600); err != nil {
		t.Fatal(err)
	}
	report := projectOracleNewReport("project")
	projectOracleRun(t, filepath.Join(project, "tsconfig.json"), t.TempDir(), report)
	if report.Outcome != "no_check" || report.Cold.Types.State != "unavailable" || report.Cold.Types.SHA256 != nil || report.Replay.State != "unavailable" {
		t.Fatalf("noCheck was counted as a semantic run: %+v", report)
	}
}

func TestProjectOracleOnDiskReplay(t *testing.T) {
	project := t.TempDir()
	config := `{"compilerOptions":{"strict":true,"noEmit":true,"skipLibCheck":true},"files":["second.ts","first.ts","ambient.d.ts"]}`
	inputs := map[string]string{
		"tsconfig.json": config,
		"second.ts": "export const second: number = 2;\n",
		"first.ts": "// @ts-expect-error\nexport const first: number = 'bad';\n",
		"ambient.d.ts": "declare const projectAmbient: string;\n",
	}
	for name, content := range inputs {
		if err := os.WriteFile(filepath.Join(project, name), []byte(content), 0o600); err != nil {
			t.Fatal(err)
		}
	}
	report := projectOracleNewReport("project")
	projectOracleRun(t, filepath.Join(project, "tsconfig.json"), t.TempDir(), report)
	if report.Outcome != "complete" {
		t.Fatalf("unexpected outcome %s: %v", report.Outcome, report.Failure)
	}
	want := []string{filepath.Join(project, "second.ts"), filepath.Join(project, "first.ts"), filepath.Join(project, "ambient.d.ts")}
	if !slices.Equal(report.Graph.ArtifactOrder, want) {
		t.Fatalf("artifact order: %v", report.Graph.ArtifactOrder)
	}
	if len(report.Cold.Diagnostics) != 0 || report.Cold.Errors.State != "no_content" {
		t.Fatal("Program comment suppression was not retained")
	}
	if len(report.Replay.Sources) != 2 || report.Cold.Walk.TypeQueries == 0 || !report.Replay.TypeIDsEqual || !report.Replay.SymbolIDsEqual {
		t.Fatalf("invalid replay evidence: %+v", report.Replay)
	}
	if len(report.Graph.Sources) <= len(want) {
		t.Fatal("the graph omitted embedded default libraries")
	}
	if !strings.Contains(report.Graph.ArtifactOrder[2], "ambient.d.ts") {
		t.Fatal("the artifact list omitted a declaration input")
	}
}
