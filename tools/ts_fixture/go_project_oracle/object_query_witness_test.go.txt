func objectWitnessFixture(t *testing.T, source, extension string) (*checker.Checker, *ast.Node) {
	t.Helper()
	filename := "/input." + extension
	config := `{"compilerOptions":{"strict":true,"skipLibCheck":true,"noEmit":true,"jsx":"preserve","types":[]},"files":["` + filename + `"]}`
	fs := bundled.WrapFS(vfstest.FromMap(map[string]string{filename: source, "/tsconfig.json": config}, true))
	host := compiler.NewCompilerHost("/", fs, bundled.LibPath(), nil, nil)
	parsed, diagnostics := tsoptions.GetParsedCommandLineOfConfigFile("/tsconfig.json", nil, nil, host, nil)
	if parsed == nil || len(diagnostics) != 0 {
		t.Fatalf("invalid fixture config: %v", diagnostics)
	}
	program := compiler.NewProgram(compiler.ProgramOptions{Host: host, Config: parsed, SingleThreaded: core.TSTrue})
	if diagnostics := program.GetSemanticDiagnostics(t.Context(), nil); len(diagnostics) != 0 {
		t.Fatalf("fixture has unexpected source diagnostics: %v", diagnostics)
	}
	file := program.GetSourceFile(filename)
	fileChecker, done := program.GetTypeCheckerForFileExclusive(t.Context(), file)
	t.Cleanup(done)
	var object *ast.Node
	var findObject func(*ast.Node) bool
	findObject = func(node *ast.Node) bool {
		if ast.IsVariableDeclaration(node) && node.Name() != nil && ast.IsIdentifier(node.Name()) && node.Name().Text() == "object" {
			object = node.Initializer()
			return object != nil
		}
		return node.ForEachChild(findObject)
	}
	file.AsNode().ForEachChild(findObject)
	if object != nil {
		return fileChecker, object
	}
	t.Fatal("fixture has no object initializer")
	return nil, nil
}

func objectWitnessObserve(t *testing.T, c *checker.Checker, node *ast.Node, mutation string) (*checker.Type, *checker.ProjectOracleObjectQueryWitness, string) {
	t.Helper()
	pass, err := c.ProjectOracleBeginObjectPass()
	if err != nil {
		t.Fatal(err)
	}
	typ, witness, err := c.ProjectOracleObjectQuery(pass, node)
	if err != nil {
		t.Fatal(err)
	}
	text := ""
	if mutation == "cycle" {
		text = c.TypeToString(typ)
	}
	if mutation != "" {
		if err := c.ProjectOracleObjectWitnessTestMutate(witness, mutation); err != nil {
			t.Fatal(err)
		}
	}
	if mutation != "cycle" {
		text = c.TypeToString(typ)
	}
	if err := c.ProjectOracleEndObjectPass(pass); err != nil {
		t.Fatal(err)
	}
	return typ, witness, text
}

func objectWitnessLog(t *testing.T, result checker.ProjectOracleObjectVerification) {
	t.Helper()
	data, err := json.Marshal(result)
	if err != nil {
		t.Fatal(err)
	}
	t.Log(string(data))
}

func TestObjectQueryWitnessAcceptsObservedProducers(t *testing.T) {
	for _, test := range []struct {
		name, source   string
		objects, edges int
	}{
		{"empty", `const object = {};`, 1, 0},
		{"properties", `const object = {number: 1, text: "a", truth: true, value: null};`, 1, 4},
		{"boolean_literal_cache", `const object = {truth: true as true, falsity: false as false};`, 1, 2},
		{"nested", `const object = {child: {value: 1}, other: {}};`, 3, 1},
		{"parenthesized", `const object = ({child: ({value: 1})});`, 2, 1},
		{"function_cache", `function stable(value: number): number { return value; } const object = {fn: stable};`, 1, 1},
		{"boolean_function_cache", `function stable(value: boolean): boolean { return value; } const object = {fn: stable};`, 1, 1},
		{"shorthand_function", `function stable(value: number): number { return value; } const object = {stable};`, 1, 1},
		{"any_cache", `const object = {value: 1 as any};`, 1, 1},
	} {
		t.Run(test.name, func(t *testing.T) {
			c, node := objectWitnessFixture(t, test.source, "ts")
			_, cold, coldText := objectWitnessObserve(t, c, node, "")
			_, warm, warmText := objectWitnessObserve(t, c, node, "")
			if coldText != warmText {
				t.Fatalf("ordinary artifact text changed: %q / %q", coldText, warmText)
			}
			result := c.ProjectOracleVerifyObjectsAfterArtifacts(cold, warm)
			objectWitnessLog(t, result)
			if result.State != "verified" || result.FreshObjects != test.objects || result.StableCacheEdges != test.edges ||
				!result.Relation.Called || result.Relation.Identical == nil || !*result.Relation.Identical ||
				result.Relation.Overflow == nil || *result.Relation.Overflow || !result.Relation.RootObserved {
				t.Fatalf("producer proof failed: %+v", result)
			}
			again := c.ProjectOracleVerifyObjectsAfterArtifacts(cold, warm)
			if again.State != "verified" || again.Relation.CachedBefore == 0 || again.Relation.Overflow == nil || *again.Relation.Overflow {
				t.Fatalf("cached relation proof failed: %+v", again)
			}
		})
	}
}

	for _, mutation := range []string{"owner", "flags", "members", "property_flags", "parent", "target", "declarations", "signature", "error_intrinsic"} {

				t.Fatalf("mutation must keep printed text equal: %q / %q", coldText, warmText)
			}

				t.Fatalf("accepted corrupted producer data: %+v", result)
			}

				anyText, errorText, identical := c.ProjectOracleObjectWitnessTestAnyError()
				if anyText != errorText || !identical {

				}
			}
		})
	}
}

func TestObjectQueryWitnessAcceptsDeclaredClassTypeParameters(t *testing.T) {
	for _, test := range []struct {
		name, source string
		edges        int
	}{
		{"call_result", `class MatchExpression<input, output> {
  make(handler: (value: input) => output, value: input) {
    const object = {matched: true as const, value: handler(value)};
    return object;
  }
}`, 2},
		{"two_parameters", `class Pair<Left, Right> {
  make(left: () => Left, right: () => Right) {
    const object = {left: left(), right: right()};
    return object;
  }
}`, 2},
	} {
		t.Run(test.name, func(t *testing.T) {
			c, node := objectWitnessFixture(t, test.source, "ts")
			coldType, cold, coldText := objectWitnessObserve(t, c, node, "")
			warmType, warm, warmText := objectWitnessObserve(t, c, node, "")
			t.Logf("raw_pointer_equal=%v cold=%q warm=%q", coldType == warmType, coldText, warmText)
			result := c.ProjectOracleVerifyObjectsAfterArtifacts(cold, warm)
			objectWitnessLog(t, result)
			if coldType == warmType || coldText != warmText {
				t.Fatalf("class parameter control requires different raw pointers and equal text: pointer_equal=%v cold=%q warm=%q", coldType == warmType, coldText, warmText)
			}
			if result.State != "verified" || result.FreshObjects != 1 || result.StableCacheEdges != test.edges ||
				!result.Relation.Called || result.Relation.Identical == nil || !*result.Relation.Identical ||
				result.Relation.Overflow == nil || *result.Relation.Overflow || result.Relation.WitnessDataChanged {
				t.Fatalf("declared class parameter cache was not proved: %+v", result)
			}
		})
	}
}

func TestObjectQueryWitnessRejectsColdWindowMutations(t *testing.T) {
	for _, mutation := range []string{
		"boolean_regular_value", "boolean_false_value", "boolean_regular_header", "boolean_regular_id", "boolean_regular_links",
		"boolean_fresh_value", "boolean_fresh_owner", "boolean_fresh_links",
		"boolean_union_constraint", "boolean_union_regular", "boolean_union_without_abstract",
		"boolean_union_property_cache", "boolean_union_constituent_map",
		"signature", "signature_and_cache", "function_base_constraint", "function_instantiations", "function_without_abstract",
		"type_parameter_cache_owner", "type_parameter_constraint", "type_parameter_base_constraint", "type_parameter_default",
		"type_parameter_target", "type_parameter_mapper", "type_parameter_owner", "type_parameter_parent",
		"type_parameter_declaration_constraint", "type_parameter_class_order", "type_parameter_memo_flags",
	} {
		windows := []string{"before_cold_close", "after_cold_close", "after_both_closes"}
		if mutation == "boolean_union_regular" {
			// This cache changes later queries, so first render both passes.
			windows = []string{"after_both_closes"}
		} else if mutation == "type_parameter_memo_flags" {
			// A later query can restore memo flags before its creation snapshot.
			windows = []string{"before_cold_close", "after_both_closes"}
		}
		for _, window := range windows {
			t.Run(mutation+"/"+window, func(t *testing.T) {
				source := `function stable(value: number): number { return value; } const object = {fn: stable};`
				if strings.HasPrefix(mutation, "boolean_") {
					source = `const object = {truth: true};`
				} else if strings.HasPrefix(mutation, "type_parameter_") {
					source = objectWitnessClassParameterSource
				}
				c, node := objectWitnessFixture(t, source, "ts")
				pass, err := c.ProjectOracleBeginObjectPass()
				if err != nil {
					t.Fatal(err)
				}
				typ, cold, err := c.ProjectOracleObjectQuery(pass, node)
				if err != nil {
					t.Fatal(err)
				}
				coldText := c.TypeToString(typ)
				mutate := func() {
					t.Helper()
					if err := c.ProjectOracleObjectWitnessTestMutate(cold, mutation); err != nil {
						t.Fatal(err)
					}
				}
				if window == "before_cold_close" {
					mutate()
				}
				if err := c.ProjectOracleEndObjectPass(pass); err != nil {
					t.Fatal(err)
				}
				if window == "after_cold_close" {
					mutate()
				}
				_, warm, warmText := objectWitnessObserve(t, c, node, "")
				if window == "after_both_closes" {
					mutate()
				}
				if coldText != warmText {
					t.Fatalf("mutation must keep artifact text equal: %q / %q", coldText, warmText)
				}
				result := c.ProjectOracleVerifyObjectsAfterArtifacts(cold, warm)
				objectWitnessLog(t, result)
				if result.State != "rejected" || result.Relation.Called {
					t.Fatalf("accepted cache data changed during %s: %+v", window, result)
				}
			})
		}
	}
}

func TestObjectQueryWitnessRejectsCopiedPassState(t *testing.T) {
	c, node := objectWitnessFixture(t, `const object = {};`, "ts")
	old, err := c.ProjectOracleBeginObjectPass()
	if err != nil {
		t.Fatal(err)
	}
	oldOpen := *old
	if err := c.ProjectOracleEndObjectPass(old); err != nil {
		t.Fatal(err)
	}
	current, err := c.ProjectOracleBeginObjectPass()
	if err != nil {
		t.Fatal(err)
	}
	registered := *current
	*current = oldOpen
	if _, _, err := c.ProjectOracleObjectQuery(current, node); err == nil {
		t.Fatal("copied pass state started a query")
	}
	if err := c.ProjectOracleEndObjectPass(current); err == nil {
		t.Fatal("copied pass state replaced its registered token")
	}
	*current = registered
	if err := c.ProjectOracleEndObjectPass(current); err != nil {
		t.Fatal(err)
	}
	missing := &checker.ProjectOracleObjectQueryWitness{}
	if result := c.ProjectOracleVerifyObjectsAfterArtifacts(missing, missing); result.State != "incomplete" || result.Relation.Called {
		t.Fatalf("missing query data was not incomplete: %+v", result)
	}
}

		t.Fatalf("cached overflow was treated as a normal relation result: %+v", result)
	}
