// These mutation helpers exist only in the added checker test overlay.
func (c *Checker) ProjectOracleObjectWitnessTestMutate(query *ProjectOracleObjectQueryWitness, change string) error {
	production := query.productions[query.result]
	if production == nil {
		return fmt.Errorf("test query has no root producer")
	}
	data := query.result.data.(*ObjectType)
	var property *ast.Symbol
	if len(data.properties) != 0 {
		property = data.properties[0]
	}
	switch change {
	case "owner":
		original := query.result.symbol
		query.result.symbol = &ast.Symbol{Flags: original.Flags, Name: original.Name,
			Declarations: slices.Clone(original.Declarations), ValueDeclaration: original.ValueDeclaration,
			Members: original.Members, Parent: original.Parent}
	case "flags":
		query.result.objectFlags &^= ObjectFlagsFreshLiteral
	case "members":
		members := make(ast.SymbolTable, len(data.members)+1)
		for name, symbol := range data.members {
			members[name] = symbol
		}
		members["unprinted"] = c.newSymbol(ast.SymbolFlagsTypeAlias, "unprinted")
		data.members = members
	case "property_flags":
		property.Flags |= ast.SymbolFlagsAssignment
	case "parent":
		property.Parent = &ast.Symbol{Flags: ast.SymbolFlagsObjectLiteral, Name: ast.InternalSymbolNameObject}
	case "target":
		target := c.valueSymbolLinks.Get(property).target
		c.valueSymbolLinks.Get(property).target = &ast.Symbol{Flags: target.Flags, Name: target.Name,
			Declarations: slices.Clone(target.Declarations), ValueDeclaration: target.ValueDeclaration, Parent: target.Parent}
	case "declarations":
		property.Declarations = nil
	case "signature", "signature_and_cache":
		function := c.valueSymbolLinks.Get(property).resolvedType.AsObjectType()
		original := function.signatures[0]
		copy := *original
		c.SignatureCount++
		copy.id = SignatureId(c.SignatureCount)
		function.signatures = []*Signature{&copy}
		if change == "signature_and_cache" {
			c.signatureLinks.Get(copy.declaration).resolvedSignature = &copy
		}
	case "signature_cache_missing":
		function := c.valueSymbolLinks.Get(property).resolvedType.AsObjectType()
		c.signatureLinks.Get(function.signatures[0].declaration).resolvedSignature = nil
	case "signature_return_unresolved":
		c.valueSymbolLinks.Get(property).resolvedType.AsObjectType().signatures[0].resolvedReturnType = nil
	case "predicate_unresolved":
		c.valueSymbolLinks.Get(property).resolvedType.AsObjectType().signatures[0].resolvedTypePredicate = nil
	case "function_base_constraint":
		c.valueSymbolLinks.Get(property).resolvedType.AsObjectType().resolvedBaseConstraint = c.numberType
	case "function_instantiations":
		c.valueSymbolLinks.Get(property).resolvedType.AsObjectType().instantiations = map[CacheHashKey]*Type{{}: c.numberType}
	case "function_without_abstract":
		c.valueSymbolLinks.Get(property).resolvedType.AsObjectType().objectTypeWithoutAbstractConstructSignatures = c.numberType
	case "type_parameter_cache_missing", "type_parameter_cache_restore", "type_parameter_cache_owner":
		typ := c.valueSymbolLinks.Get(property).resolvedType
		cached := typ
		if change == "type_parameter_cache_missing" {
			cached = nil
		} else if change == "type_parameter_cache_owner" {
			cached = c.numberType
		}
		c.declaredTypeLinks.Get(typ.symbol).declaredType = cached
	case "type_parameter_constraint":
		c.valueSymbolLinks.Get(property).resolvedType.AsTypeParameter().constraint = c.stringType
	case "type_parameter_base_constraint":
		c.valueSymbolLinks.Get(property).resolvedType.AsTypeParameter().resolvedBaseConstraint = c.stringType
	case "type_parameter_default":
		c.valueSymbolLinks.Get(property).resolvedType.AsTypeParameter().resolvedDefaultType = c.stringType
	case "type_parameter_target":
		typ := c.valueSymbolLinks.Get(property).resolvedType
		other := typ.symbol.Declarations[0].Parent.TypeParameters()[1].Symbol()
		links := c.declaredTypeLinks.TryGet(other)
		if links == nil || links.declaredType == nil {
			return fmt.Errorf("test has no second declared class parameter")
		}
		typ.AsTypeParameter().target = links.declaredType
	case "type_parameter_mapper":
		c.valueSymbolLinks.Get(property).resolvedType.AsTypeParameter().mapper = &TypeMapper{}
	case "type_parameter_memo_flags":
		c.valueSymbolLinks.Get(property).resolvedType.objectFlags ^= ObjectFlagsCouldContainTypeVariablesComputed | ObjectFlagsCouldContainTypeVariables
	case "type_parameter_owner":
		typ := c.valueSymbolLinks.Get(property).resolvedType
		original := typ.symbol
		typ.symbol = &ast.Symbol{Flags: original.Flags, Name: original.Name, Parent: original.Parent,
			Declarations: slices.Clone(original.Declarations), ValueDeclaration: original.ValueDeclaration}
	case "type_parameter_parent":
		declaration := c.valueSymbolLinks.Get(property).resolvedType.symbol.Declarations[0]
		for _, statement := range query.file.Statements.Nodes {
			if ast.IsClassDeclaration(statement) && statement.Name() != nil && statement.Name().Text() == "OtherOwner" {
				declaration.Parent = statement
				return nil
			}
		}
		return fmt.Errorf("test has no alternate class owner")
	case "type_parameter_declaration_constraint":
		declaration := c.valueSymbolLinks.Get(property).resolvedType.symbol.Declarations[0]
		declaration.AsTypeParameterDeclaration().Constraint = declaration.Name()
	case "type_parameter_class_order":
		parent := c.valueSymbolLinks.Get(property).resolvedType.symbol.Declarations[0].Parent.AsClassDeclaration()
		nodes := slices.Clone(parent.TypeParameters.Nodes)
		nodes[0], nodes[1] = nodes[1], nodes[0]
		parent.TypeParameters.Nodes = nodes
	case "boolean_regular_value":
		c.regularTrueType.AsLiteralType().value = false
	case "boolean_false_value":
		c.regularFalseType.AsLiteralType().value = true
	case "boolean_regular_header":
		c.regularTrueType.objectFlags |= ObjectFlagsNonInferrableType
	case "boolean_regular_id":
		c.regularTrueType.id++
	case "boolean_regular_links":
		c.regularTrueType.AsLiteralType().regularType = c.regularFalseType
	case "boolean_fresh_value":
		c.trueType.AsLiteralType().value = false
	case "boolean_fresh_owner":
		c.trueType.checker = nil
	case "boolean_fresh_links":
		c.falseType.AsLiteralType().freshType = c.trueType
	case "boolean_union_constraint":
		c.booleanType.AsUnionType().resolvedBaseConstraint = c.numberType
	case "boolean_union_regular":
		c.booleanType.AsUnionType().regularType = c.numberType
	case "boolean_union_without_abstract":
		c.booleanType.AsUnionType().objectTypeWithoutAbstractConstructSignatures = c.numberType
	case "boolean_union_property_cache":
		c.booleanType.AsUnionType().propertyCache = ast.SymbolTable{"hidden": c.newProperty("hidden", c.numberType)}
	case "boolean_union_constituent_map":
		c.booleanType.AsUnionType().constituentMap = map[*Type]*Type{c.regularTrueType: c.numberType}
	case "error_intrinsic":
		c.valueSymbolLinks.Get(property).resolvedType = c.errorType
	case "query_token":
		query.token++
	case "producer_token":
		production.queryToken++
	case "cycle":
		c.valueSymbolLinks.Get(property).resolvedType = query.result
		for index := range production.birth.properties {
			if production.birth.properties[index].symbol.ref == property {
				production.birth.properties[index].links.resolvedType = query.result
			}
		}
	default:
		return fmt.Errorf("unknown test mutation %q", change)
	}
	return nil
}

func (c *Checker) ProjectOracleObjectWitnessTestCacheEvidence(query *ProjectOracleObjectQueryWitness) (bool, bool) {
	production := query.productions[query.result]
	property := query.result.AsObjectType().properties[0]
	_, birth := production.birth.cacheEdges[property]
	_, closed := query.closed[query.result].cacheEdges[property]
	return birth, closed
}

func (c *Checker) ProjectOracleObjectWitnessTestMissingClassParameterCache(query *ProjectOracleObjectQueryWitness) (known, allocated, cacheCreated bool) {
	property := query.result.AsObjectType().properties[0]
	typ := c.valueSymbolLinks.TryGet(property).resolvedType
	before := c.declaredTypeLinks
	defer func() { c.declaredTypeLinks = before }()
	c.declaredTypeLinks = core.LinkStore[*ast.Symbol, DeclaredTypeLinks]{}
	types, symbols, signatures := c.TypeCount, c.SymbolCount, c.SignatureCount
	_, known = c.objectWitnessClassTypeParameterSnapshot(typ)
	return known, types != c.TypeCount || symbols != c.SymbolCount || signatures != c.SignatureCount, c.declaredTypeLinks.Has(typ.symbol)
}

func (c *Checker) ProjectOracleObjectWitnessTestSeedFlow(node *ast.Node, typ *Type) {
	if c.flowTypeCache == nil {
		c.flowTypeCache = map[*ast.Node]*Type{}
	}
	c.flowTypeCache[node] = typ
}

func (c *Checker) ProjectOracleObjectWitnessTestSeedExpando(node *ast.Node) {
	node.Symbol().Exports = ast.SymbolTable{"value": c.newProperty("value", c.numberType)}
}

func (c *Checker) ProjectOracleObjectWitnessTestSeedOverflow(left, right *ProjectOracleObjectQueryWitness) {
	key, _ := getRelationKey(left.result, right.result, IntersectionStateNone, true, false)
	c.identityRelation.set(key, RelationComparisonResultFailed|RelationComparisonResultStackDepthOverflow)
}

func (c *Checker) ProjectOracleObjectWitnessTestAnyError() (string, string, bool) {
	return c.TypeToString(c.anyType), c.TypeToString(c.errorType), c.isTypeIdenticalTo(c.anyType, c.errorType)
}
