func runPhaseOracleControl(t *testing.T, source string) (*phaseOracleReport, []phaseOracleEvent) {
	t.Helper()
	config := projectOracleTestConfig(t, `{"compilerOptions":{"strict":true,"declaration":true,"skipLibCheck":true,"noUnusedLocals":true,"noUnusedParameters":true,"target":"ES2022","module":"ES2020","moduleResolution":"Bundler","types":[]},"files":["input.ts"]}`, map[string]string{"input.ts": source})
	output := t.TempDir()
	report := newPhaseOracleReport(config)
	recorder := newPhaseOracleRecorder(output)
	phaseOracleRun(t, config, output, report, recorder)
	report.Trace = recorder.close()
	report.CheckerBindings = recorder.bindings
	traceBytes, err := os.ReadFile(filepath.Join(output, report.Trace.Path))
	if err != nil {
		t.Fatal(err)
	}
	if projectOracleHash(traceBytes) != report.Trace.SHA256 {
		t.Fatal("trace hash differs")
	}
	file, err := os.Open(filepath.Join(output, report.Trace.Path))
	if err != nil {
		t.Fatal(err)
	}
	defer file.Close()
	scanner := bufio.NewScanner(file)
	events := []phaseOracleEvent{}
	for scanner.Scan() {
		var event phaseOracleEvent
		if err := json.Unmarshal(scanner.Bytes(), &event); err != nil {
			t.Fatal(err)
		}
		events = append(events, event)
	}
	if err := scanner.Err(); err != nil {
		t.Fatal(err)
	}
	return report, events
}

func findPhaseSnapshot(t *testing.T, p *phaseOracleProgramReport, name string) phaseOracleSnapshot {
	t.Helper()
	for _, snapshot := range p.Snapshots {
		if snapshot.Phase == name {
			return snapshot
		}
	}
	t.Fatalf("missing phase %s", name)
	return phaseOracleSnapshot{}
}

func TestPhaseOracleFrozenFalseAndSourceOnly(t *testing.T) {
	source := `export namespace JSX { export interface IntrinsicElements { div?: string; } }
export interface IntrinsicElements extends JSX.IntrinsicElements {}
export function makeCache(): () => Promise<number> {
  return async function cache() { const cache = 1; return cache; };
}
export function makeEtag(): () => Promise<string | null> {
  return async function etag() { let etag: string | null = null; return etag; };
}`
	report, events := runPhaseOracleControl(t, source)
	if report.SchemaVersion != 3 || len(report.Programs) != 2 {
		t.Fatal("missing versioned independent Programs")
	}
	plain, queried := report.Programs[0], report.Programs[1]
	if plain.Token == queried.Token || plain.Checker == queried.Checker {
		t.Fatal("Programs or checkers were shared")
	}
	if plain.ArtifactQueries != 0 || len(plain.ArtifactWalks) != 0 {
		t.Fatal("source-only Program had artifact queries")
	}
	for _, snapshot := range plain.Snapshots {
		if len(snapshot.Diagnostics) != 0 {
			t.Fatalf("source-only diagnostics changed: %+v", snapshot)
		}
	}
	if len(findPhaseSnapshot(t, queried, "initial_source").Diagnostics) != 0 {
		t.Fatal("initial source snapshot is not clean")
	}
	if len(findPhaseSnapshot(t, queried, "first_before_types").Diagnostics) != 0 {
		t.Fatal("before-types snapshot is not clean")
	}
	for _, name := range []string{"first_after_types", "first_after_symbols", "before_reset", "after_reentry", "second_before_types", "second_after_types", "second_after_symbols"} {
		snapshot := findPhaseSnapshot(t, queried, name)
		if len(snapshot.Diagnostics) != 5 || len(snapshot.DeltaFromInitial.Added) != 5 {
			t.Fatalf("query-added diagnostics not isolated at %s: %+v", name, snapshot)
		}
	}
	for _, walk := range queried.ArtifactWalks {
		if walk.HadErrorBaseline || walk.SourceSnapshot != queried.InitialSourceSnapshot {
			t.Fatal("renderer flag was not frozen from the initial clean source snapshot")
		}
	}
	if report.FreshDiagnostics.State != "unavailable" || report.FreshDiagnostics.Equal != nil || report.Outcome == "complete" {
		t.Fatal("retained evidence became fresh parity")
	}
	var sequence, queries uint64
	bindings := map[string]string{}
	for _, event := range events {
		sequence++
		if event.Sequence != sequence || event.Token != fmt.Sprintf("operation-%08d", sequence) {
			t.Fatal("operation sequence is not exact")
		}
		if event.Operation == "checker_bound" {
			bindings[event.Checker] = event.Program
		}
		if event.Checker != "" && bindings[event.Checker] != event.Program {
			t.Fatal("checker token was not bound to its Program")
		}
		if event.Query != "" {
			queries++
			if event.Query != fmt.Sprintf("query-%08d", queries) || event.Program != queried.Token || event.Checker != queried.Checker {
				t.Fatal("query tokens crossed Programs or checkers")
			}
			if event.Queried == nil || event.Rendered == nil {
				t.Fatal("query trace omitted its exact nodes")
			}
		}
	}
	if sequence != report.Trace.Events || queries != report.Trace.Queries || queries != queried.ArtifactQueries {
		t.Fatal("trace totals differ")
	}
}

func TestPhaseOracleFrozenTrue(t *testing.T) {
	report, _ := runPhaseOracleControl(t, `export const value: number = "bad";`)
	for _, p := range report.Programs {
		initial := findPhaseSnapshot(t, p, "initial_source")
		if !slices.ContainsFunc(initial.Diagnostics, func(d projectOracleDiagnostic) bool { return d.Code == 2322 }) {
			t.Fatal("initial source diagnostic missing")
		}
	}
	for _, walk := range report.Programs[1].ArtifactWalks {
		if !walk.HadErrorBaseline {
			t.Fatal("initial error flag was lost")
		}
	}
	if report.Outcome != "incomplete_evidence" || report.FreshDiagnostics.Equal != nil {
		t.Fatalf("unexpected proof state: %s", report.Outcome)
	}
}

func TestPhaseOracleKeepsStrictPointerGate(t *testing.T) {
	report, _ := runPhaseOracleControl(t, `export const empty = {};`)
	if !report.PointerChecks.TypesEqual || !report.PointerChecks.SymbolsEqual {
		t.Fatal("control artifact bytes changed")
	}
	if report.PointerChecks.TypeQueriesEqual || report.PointerChecks.RenderedTypesEqual || report.PointerChecks.ChangedTypeQueries == 0 {
		t.Fatal("fresh object query identity change was hidden")
	}
	if report.Outcome != "invariant_error" {
		t.Fatal("equal text bypassed the strict pointer gate")
	}
}
