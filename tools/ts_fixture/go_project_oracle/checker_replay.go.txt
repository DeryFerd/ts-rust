package checker

import "github.com/microsoft/typescript-go/internal/ast"

// ProjectOracleResetSource clears completion flags without replacing cached types.
// The caller must hold the Program's exclusive checker lock.
func (c *Checker) ProjectOracleResetSource(file *ast.SourceFile) (typeChecked, unusedChecked bool) {
	links := c.sourceFileLinks.Get(file)
	typeChecked, unusedChecked = links.typeChecked, links.unusedChecked
	links.typeChecked = false
	links.unusedChecked = false
	return
}
